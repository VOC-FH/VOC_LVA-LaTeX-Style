\def\definejavacoderangeenvironment#1#2#3{%  #1 startlinenumber, #2 = linenumbercount #3 javafile
  \newenvironment{#3}[1][]{% #1 globaloptions
    \begingroup%
      \offinterlineskip%
      \def\@globalargs{##1}%
      \expandafter\def\expandafter\range\expandafter{\csname javarange#3\endcsname}%
      \csname javarel#3\endcsname%
  }{
    \@endcmd\newline%
    \endgroup%
    \ignorespacesafterend
  }%
%
  \expandafter\newcommand\expandafter{\csname java#3\endcsname}[1][#2]{%  ##1 = endoptions
    \edef\@endoptions{[linerange=##1-#2,firstnumber=##1,\@globalargs]{#3}}%
    \def\@endcmd{\expandafter\includejavacode\@endoptions}%
    \edef\@startoptions{[\@startendlinenumber,caption=,belowskip=-0.5ex,\@globalargs]{#3}}%
    \def\@startcmd{\expandafter\includejavacode\@startoptions}%
    \@startcmd\newline%
    \ignorespaces%
  }%
%
  \expandafter\newcommand\expandafter{\csname javarel#3\endcsname}[1][#1]{% ##1 = startargs
    \def\@startendlinenumber{linerange=#1-##1}%
    \expandafter\csname java#3\expandafter\endcsname%
  }%
%
  \expandafter\newcommand\expandafter{\csname javarange#3\endcsname}[3][]{% ##1 = rangeoptions, ##2 = startline, ##3 = endline
    \edef\@rangeoptions{[linerange=##2-##3,firstnumber=##2,belowskip=-0.5ex,caption=,\@globalargs,##1]{#3}}%
    \expandafter\includejavacode\@rangeoptions\newline%
    \ignorespaces
  }%
}%

\def\definecppcoderangeenvironment#1{%  #1 startlinenumber, #2 = linenumbercount #3 javafile
  \newenvironment{#1}[1][]{% #1 globaloptions
    \begingroup%
      \offinterlineskip%
      \def\@globalargs{##1}%
      \expandafter\def\expandafter\range\expandafter{\csname cpprange#1\endcsname}%
  }{
    \endgroup%
    \ignorespacesafterend
  }%

  \expandafter\newcommand\expandafter{\csname cpprange#1\endcsname}[3][]{% ##1 = rangeoptions, ##2 = startline, ##3 = endline
    \edef\@rangeoptions{[linerange=##2-##3,firstnumber=##2,belowskip=-0.5ex,\@globalargs,##1]{#1}}%
    \expandafter\includecppcode\@rangeoptions\newline%
    \ignorespaces
  }%
}%
